import { Readable } from "stream";

function tapStream(stream: Readable, onDone: (info: any) => void) {
  let bytes = 0;
  let first: Buffer | null = null;

  stream.on("data", (chunk: Buffer) => {
    bytes += chunk.length;
    if (!first) first = chunk.slice(0, 16);
  });
  stream.on("end", () => onDone({ bytes, firstHex: first?.toString("hex") || null, event: "end" }));
  stream.on("close", () => onDone({ bytes, firstHex: first?.toString("hex") || null, event: "close" }));
  stream.on("error", (e) => onDone({ bytes, firstHex: first?.toString("hex") || null, event: "error", err: String(e) }));
  return stream;
}

export async function PUT(req: NextRequest) {
  ...
  if (!req.body) throw new Error("Missing req.body");

  const nodeStream = Readable.fromWeb(req.body as any);
  const tapped = tapStream(nodeStream, (info) => console.log("[upload] stream", info));

  await objectStorage.uploadFromStream(objectKey, tapped);
  ...
}
