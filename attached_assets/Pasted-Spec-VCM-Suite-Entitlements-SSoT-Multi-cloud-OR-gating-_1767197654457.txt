const TIER_ORDER: EntitlementTier[] = ['free', 'basic', 'pro', 'enterprise'];

export function hasAccess(
  entitlements: CloudEntitlement[] | undefined,
  cloud: CloudId,
  required: EntitlementTier
): boolean {
  const record = entitlements?.find(e => e.cloud_id === cloud && (e.status === 'active' || e.status === 'trial'));
  if (!record) return false;
  return TIER_ORDER.indexOf(record.tier) >= TIER_ORDER.indexOf(required);
}
Add gateTool() (the only gate anyone calls)
Create lib/entitlements/gateTool.ts:
export type GateReason =
  | 'allowed'
  | 'not_logged_in'
  | 'no_entitlement'
  | 'tier_too_low'
  | 'misconfigured_tool';

export interface GateResult {
  allowed: boolean;
  reason: GateReason;

  required_tier: EntitlementTier;
  user_tier_best: EntitlementTier | null;

  // For UX / upgrade
  upgrade_cloud_id: CloudId | null;

  // Debug
  checked_cloud_ids: CloudId[];
}

export function gateTool(params: {
  tool: Pick<Tool, 'cloud_ids' | 'required_tier'>;
  entitlements: CloudEntitlement[] | undefined;
  isLoggedIn: boolean;
}): GateResult {
  const { tool, entitlements, isLoggedIn } = params;

  // Misconfig protection
  if (!tool.cloud_ids || tool.cloud_ids.length === 0) {
    return {
      allowed: tool.required_tier === 'free', // free tools can be accessible even if untagged (optional)
      reason: tool.required_tier === 'free' ? 'allowed' : 'misconfigured_tool',
      required_tier: tool.required_tier,
      user_tier_best: null,
      upgrade_cloud_id: null,
      checked_cloud_ids: tool.cloud_ids ?? []
    };
  }

  // Free tier tools are always allowed
  if (tool.required_tier === 'free') {
    return {
      allowed: true,
      reason: 'allowed',
      required_tier: tool.required_tier,
      user_tier_best: null,
      upgrade_cloud_id: null,
      checked_cloud_ids: tool.cloud_ids
    };
  }

  if (!isLoggedIn) {
    return {
      allowed: false,
      reason: 'not_logged_in',
      required_tier: tool.required_tier,
      user_tier_best: null,
      upgrade_cloud_id: tool.cloud_ids[0] ?? null,
      checked_cloud_ids: tool.cloud_ids
    };
  }

  // OR logic across clouds
  const ok = tool.cloud_ids.some(cloudId => hasAccess(entitlements, cloudId, tool.required_tier));

  if (ok) {
    // optional: compute best tier user has across the tool clouds for display
    const active = entitlements?.filter(e => tool.cloud_ids.includes(e.cloud_id) && (e.status === 'active' || e.status === 'trial')) ?? [];
    const best = active
      .map(e => e.tier)
      .sort((a, b) => TIER_ORDER.indexOf(b) - TIER_ORDER.indexOf(a))[0] ?? null;

    return {
      allowed: true,
      reason: 'allowed',
      required_tier: tool.required_tier,
      user_tier_best: best,
      upgrade_cloud_id: null,
      checked_cloud_ids: tool.cloud_ids
    };
  }

  // Decide best upsell target (simple v1: first cloud)
  return {
    allowed: false,
    reason: entitlements?.some(e => tool.cloud_ids.includes(e.cloud_id) && (e.status === 'active' || e.status === 'trial'))
      ? 'tier_too_low'
      : 'no_entitlement',
    required_tier: tool.required_tier,
    user_tier_best: 'free',
    upgrade_cloud_id: tool.cloud_ids[0] ?? null,
    checked_cloud_ids: tool.cloud_ids
  };
}

export function withToolGate(handler) {
  return async (req, res) => {
    const tool = await getToolFromReq(req); // slug param
    const session = await getSession(req);
    const entitlements = session ? await getCloudEntitlementsForUser(session.userId) : undefined;

    const gate = gateTool({ tool, entitlements, isLoggedIn: !!session });

    if (!gate.allowed) {
      return res.status(403).json({ ok: false, gate });
    }

    return handler(req, res, { tool, session, entitlements });
  };
}
