3) Replace finalize with a dead-simple implementation (no wrappers)
This removes 90% of “I think” debugging.
import { PassThrough } from "stream";

export async function finalize(jobId: string, partKeys: string[], finalKey: string) {
  const pass = new PassThrough();

  const uploadPromise = client.uploadFromStream(finalKey, pass);

  let total = 0;

  for (let i = 0; i < partKeys.length; i++) {
    const key = partKeys[i];
    const buf = await client.downloadAsBytes(key);

    if (!Buffer.isBuffer(buf)) throw new Error(`[finalize] not a buffer for ${key}`);
    if (buf.length === 0) throw new Error(`[finalize] empty buffer for ${key}`);

    console.log(`[finalize] part ${i+1}/${partKeys.length} len=${buf.length} hex16=${buf.subarray(0,16).toString("hex")}`);

    // backpressure-safe
    if (!pass.write(buf)) {
      await new Promise<void>((resolve, reject) => {
        pass.once("drain", resolve);
        pass.once("error", reject);
      });
    }

    total += buf.length;
  }

  pass.end();
  await uploadPromise;

  console.log(`[finalize] assembledBytes=${total}`);
}
This is the correct pattern. No “race conditions.” No “double resolve.” Just bytes.
Why I’m calling BS on the “write callback” theory
Because even with a broken drain wrapper, you don’t reliably end up with exactly N bytes where N = number of chunks unless:
you are literally doing total += 1 per chunk, or
you are literally writing a 1-byte buffer per chunk, or
your stored chunk objects are 1 byte each.
Those are the only three explanations consistent with your logs.
